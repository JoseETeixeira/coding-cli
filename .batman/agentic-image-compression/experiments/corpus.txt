def estimate_tokens(text: str) -> int:
    """Estimate conservatively: each ASCII run costs ceil(chars/4), non-ASCII 1."""

    tokens = 0
    ascii_run = 0
    for char in text:
        if ord(char) < 128:
            ascii_run += 1
            continue
        if ascii_run:
            tokens += ceil(ascii_run / 4)
            ascii_run = 0
        tokens += 1
    if ascii_run:
        tokens += ceil(ascii_run / 4)
    return tokens
